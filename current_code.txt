from django.db import models


class Post(models.Model):
    CATEGORY_CHOICES = [
        ("architecture", "건축"),
        ("realestate", "부동산"),
        ("finance", "금융"),
        ("tech", "테크"),
        ("life", "일상"),
    ]

    category = models.CharField(
        max_length=20,
        choices=CATEGORY_CHOICES,
        verbose_name="카테고리"
    )

    title = models.CharField(
        max_length=200,
        verbose_name="제목"
    )

    content = models.TextField(
        verbose_name="내용"
    )

    thumbnail = models.ImageField(
        upload_to="post_thumbnails/",
        blank=True,
        null=True,
        verbose_name="썸네일"
    )

    content_image = models.ImageField(
        upload_to="post_content_images/",
        blank=True,
        null=True,
        verbose_name="본문 사진"
    )

    location = models.CharField(
        max_length=200,
        blank=True,
        verbose_name="위치정보"
    )

    tags = models.CharField(
        max_length=300,
        blank=True,
        verbose_name="태그"
    )

    views = models.PositiveIntegerField(
        default=0,
        verbose_name="조회수"
    )

    created_at = models.DateTimeField(
        auto_now_add=True,
        verbose_name="등록일"
    )

    updated_at = models.DateTimeField(
        auto_now=True,
        verbose_name="수정일"
    )

    def __str__(self):
        return self.title

    @property
    def tag_list(self):
        if not self.tags:
            return []
        return [tag.strip() for tag in self.tags.split(",") if tag.strip()]

def admin_required(user):
    return user.is_authenticated and (user.is_staff or user.is_superuser)


def home(request):
    posts = Post.objects.all().order_by("-created_at")[:6]

    return render(request, "core/home.html", {
        "posts": posts,
    })


def category_page(request, slug):
    page = CATEGORY_PAGES.get(slug)

    if page is None:
        raise Http404("존재하지 않는 페이지입니다.")

    posts = Post.objects.filter(category=slug).order_by("-created_at")[:15]

    return render(request, "core/category.html", {
        "page": page,
        "slug": slug,
        "posts": posts,
    })


def search(request):
    query = request.GET.get("q", "").strip()
    results = Post.objects.none()

    category_keywords = {
        "건축": "architecture",
        "부동산": "realestate",
        "금융": "finance",
        "테크": "tech",
        "기술": "tech",
        "일상": "life",
    }

    category_slug = category_keywords.get(query)

    if query:
        search_filter = (
            Q(title__icontains=query) |
            Q(content__icontains=query) |
            Q(category__icontains=query)
        )

        if category_slug:
            search_filter = search_filter | Q(category=category_slug)

        results = Post.objects.filter(search_filter).order_by("-created_at")

    return render(request, "core/search.html", {
        "query": query,
        "results": results,
    })


def post_detail(request, pk):
    post = get_object_or_404(Post, pk=pk)

    post.views += 1
    post.save(update_fields=["views"])

    return render(request, "core/post_detail.html", {
        "post": post,
    })


@user_passes_test(admin_required)
def post_create(request):
    initial_category = request.GET.get("category", "")

    if request.method == "POST":
        form = PostForm(request.POST, request.FILES)

        if form.is_valid():
            post = form.save()
            return redirect("post_detail", pk=post.pk)
    else:
        form = PostForm(initial={
            "category": initial_category,
        })

    return render(request, "core/post_form.html", {
        "form": form,
        "mode": "create",
        "post": None,
    })


@user_passes_test(admin_required)
def post_update(request, pk):
    post = get_object_or_404(Post, pk=pk)

    if request.method == "POST":
        form = PostForm(request.POST, request.FILES, instance=post)

        if form.is_valid():
            post = form.save()
            return redirect("post_detail", pk=post.pk)
    else:
        form = PostForm(instance=post)

    return render(request, "core/post_form.html", {
        "form": form,
        "mode": "update",
        "post": post,
    })


@user_passes_test(admin_required)
def post_delete(request, pk):
    post = get_object_or_404(Post, pk=pk)
    category = post.category

    if request.method == "POST":
        post.delete()
        return redirect(category)

    return redirect("post_detail", pk=post.pk)

def signup(request):
    if request.method == "POST":
        form = UserCreationForm(request.POST)

        if form.is_valid():
            user = form.save()
            auth_login(request, user)
            return redirect("home")
    else:
        form = UserCreationForm()

    return render(request, "registration/signup.html", {
        "form": form,
    })

    function updateHomeDateTime() {

            return;

        const year = now.getFullYear();
        const month = String(now.getMonth() + 1).padStart(2, "0");
        const day = String(now.getDate()).padStart(2, "0");
        const hour = String(now.getHours()).padStart(2, "0");
        const minute = String(now.getMinutes()).padStart(2, "0");
        const second = String(now.getSeconds()).padStart(2, "0");

        dateBox.textContent = `${year}년 ${month}월 ${day}일 ${hour}:${minute}:${second}`;
    }

    updateHomeDateTime();
    setInterval(updateHomeDateTime, 1000);

<script>
    const editor = document.getElementById("editor");
    const hiddenContent = document.getElementById("id_content");
    const form = document.querySelector(".tistory-editor-form");
    const charCount = document.getElementById("content-char-count");

    const locationModal = document.getElementById("location-modal");
    const locationInput = document.getElementById("location-search-input");
    const locationResultArea = document.getElementById("location-result-area");

    function formatDoc(command) {
        editor.focus();
        document.execCommand(command, false, null);
        updateCharCount();
    }

    function formatBlock(tag) {
        if (!tag) {
            return;
        }

        editor.focus();
        document.execCommand("formatBlock", false, tag);
        updateCharCount();
    }

    function changeFontSize(size) {
        if (!size) {
            return;
        }

        editor.focus();

        const selection = window.getSelection();

        if (!selection.rangeCount) {
            return;
        }

        const range = selection.getRangeAt(0);
        const selectedText = range.extractContents();

        const span = document.createElement("span");
        span.style.fontSize = size;
        span.appendChild(selectedText);

        range.insertNode(span);
        updateCharCount();
    }

    function insertLink() {
        const url = prompt("링크 주소를 입력하세요.");

        if (!url) {
            return;
        }

        editor.focus();
        document.execCommand("createLink", false, url);
        updateCharCount();
    }

    function insertImage(event) {
        const file = event.target.files[0];

        if (!file) {
            return;
        }

        const reader = new FileReader();

        reader.onload = function(e) {
            editor.focus();

            const img = document.createElement("img");
            img.src = e.target.result;
            img.className = "editor-inserted-image";

            const wrapper = document.createElement("div");
            wrapper.className = "editor-image-wrap";
            wrapper.appendChild(img);

            insertNodeToEditor(wrapper);
            updateCharCount();
        };

        reader.readAsDataURL(file);
    }

    function openLocationModal() {
        locationModal.classList.add("active");

        locationResultArea.innerHTML = `
            <div class="location-empty-message">
                검색어를 입력해주세요.
            </div>
        `;

        setTimeout(function () {
            locationInput.focus();
        }, 50);
    }

    function closeLocationModal() {
        locationModal.classList.remove("active");
        locationInput.value = "";
    }

    function handleLocationKeydown(event) {
        if (event.key === "Enter") {
            event.preventDefault();
            performLocationSearch();
        }

        if (event.key === "Escape") {
            closeLocationModal();
        }
    }

    function performLocationSearch() {
        const keyword = locationInput.value.trim();

        if (!keyword) {
            locationResultArea.innerHTML = `
                <div class="location-empty-message">
                    검색어를 입력해주세요.
                </div>
            `;
            return;
        }

        const results = [
            {
                type: "장소",
                name: keyword,
                address: "입력한 장소명을 기준으로 지도 검색 링크를 생성합니다."
            },
            {
                type: "네이버",
                name: keyword + " 네이버지도 검색",
                address: "네이버지도에서 해당 장소를 검색합니다."
            },
            {
                type: "구글",
                name: keyword + " 구글지도 검색",
                address: "구글지도에서 해당 장소를 검색합니다."
            }
        ];

        locationResultArea.innerHTML = "";

        results.forEach(function (item) {
            const resultItem = document.createElement("button");
            resultItem.type = "button";
            resultItem.className = "location-result-item";
            resultItem.onclick = function () {
                selectLocation(item.name, item.address);
            };

            resultItem.innerHTML = `
                <span class="location-result-badge">${item.type}</span>
                <div class="location-result-text">
                    <strong>${escapeHtml(item.name)}</strong>
                    <p>${escapeHtml(item.address)}</p>
                </div>
            `;

            locationResultArea.appendChild(resultItem);
        });
    }

    function selectLocation(placeName, placeAddress) {
        const encodedPlace = encodeURIComponent(placeName);

        const naverMapUrl = `https://map.naver.com/p/search/${encodedPlace}`;
        const googleMapUrl = `https://www.google.com/maps/search/?api=1&query=${encodedPlace}`;

        const locationBox = document.createElement("div");
        locationBox.className = "editor-location-card";
        locationBox.innerHTML = `
            <div class="editor-location-icon">📍</div>
            <div class="editor-location-content">
                <strong>${escapeHtml(placeName)}</strong>
                <p>${escapeHtml(placeAddress)}</p>
                <div class="editor-location-links">
                    <a href="${naverMapUrl}" target="_blank" rel="noopener noreferrer">네이버지도</a>
                    <a href="${googleMapUrl}" target="_blank" rel="noopener noreferrer">구글지도</a>
                </div>
            </div>
        `;

        editor.focus();
        insertNodeToEditor(locationBox);
        updateCharCount();
        closeLocationModal();
    }

    function insertNodeToEditor(node) {
        const selection = window.getSelection();

        if (selection.rangeCount > 0) {
            const range = selection.getRangeAt(0);
            range.insertNode(node);

            const spacer = document.createElement("p");
            spacer.innerHTML = "<br>";
            node.after(spacer);

            range.setStartAfter(spacer);
            range.setEndAfter(spacer);
            selection.removeAllRanges();
            selection.addRange(range);
        } else {
            editor.appendChild(node);

            const spacer = document.createElement("p");
            spacer.innerHTML = "<br>";
            editor.appendChild(spacer);
        }
    }

    function escapeHtml(text) {
        const div = document.createElement("div");
        div.textContent = text;
        return div.innerHTML;
    }

    function updateCharCount() {
        if (!editor || !charCount) {
            return;
        }

        let text = editor.innerText || "";
        text = text.replace("내용을 입력하세요.", "").trim();

        const count = text.length;
        charCount.textContent = `글자수 ${count.toLocaleString()}자`;
    }

    editor.addEventListener("input", updateCharCount);
    editor.addEventListener("keyup", updateCharCount);
    editor.addEventListener("paste", function () {
        setTimeout(updateCharCount, 50);
    });

    form.addEventListener("submit", function () {
        hiddenContent.value = editor.innerHTML;
    });

    updateCharCount();
</script>
